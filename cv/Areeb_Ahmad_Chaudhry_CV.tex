\section{Research Experience}
\entry{\href{https://vision.seecs.edu.pk/}{MachVIS Lab}, SEECS, NUST}{2024 -- Sep 2025}{Research Assistant (Jun -- Sep 2025); undergraduate researcher during final-year project. Advisor: \href{https://scholar.google.com/citations?user=8q5XnWQAAAAJ}{Dr.~Nazia Perwaiz}}{Islamabad, Pakistan}

\vspace{6pt}
\textbf{FeLoRA16-SPP: 3D multimodal LLM for CT report generation and VQA} \hfill Jun -- Sep 2025
\begin{itemize}
  \item Adapted a 3D multimodal LLM (M3D-LaMed) to write reports and answer clinical questions about CT scans, training only about 8\% of its 4.07B parameters by combining LoRA with a spatial pooling projector.
  \item Reached an average GREEN score of 0.431 across 18 organ systems, up to 12\% above PHI3 and 4\% above Med3DVLM, with top scores on 9 organs; global VQA accuracy remained below the baselines.
\end{itemize}

\textbf{Radixpert: anatomy-guided VLM for radiology (Final Year Project, Grade A)} \hfill Sep 2024 -- Jun 2025
\begin{itemize}
  \item Co-developed a model that conditions a language model (MedGemma-4B) on both chest X-ray features and anatomical segmentation masks, so that generated reports are tied to anatomy; trained about 1.8\% of parameters with LoRA on 16,000 PadChest studies.
  \item Reached RadCliQ 0.804, comparable to the strongest baseline (0.810); ablations showed each component's contribution (0.682 $\rightarrow$ 0.755 $\rightarrow$ 0.804), and anatomical grounding reduced left/right errors.
\end{itemize}
